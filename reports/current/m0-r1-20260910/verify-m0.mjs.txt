import fs from 'node:fs';
import path from 'node:path';
import {spawnSync,execFileSync} from 'node:child_process';
import {cleanEnv} from '../scripts/audit/radio-lab.mjs';
const root=process.cwd();
const out=path.join(root,'tmp/final-verification');fs.mkdirSync(out,{recursive:true});
const temp=path.join(root,'tmp/process-temp');fs.mkdirSync(temp,{recursive:true});
const sha=execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim();
const env={...cleanEnv(),TEMP:temp,TMP:temp,NODE_OPTIONS:'--max-old-space-size=4096',NEXT_TELEMETRY_DISABLED:'1',npm_config_cache:path.join(root,'tmp/npm-cache')};
const cases=[
 ['lint',['node_modules/eslint/bin/eslint.js','.']],
 ['unit',['node_modules/vitest/vitest.mjs','run','--exclude=tests/**','--pool=threads','--maxWorkers=1']],
 ['ci',['--test','scripts/ci/comun-central.node-test.mjs','scripts/ci/vercel-build-impact.node-test.mjs']],
 ['ffmpeg',['C:/Users/Micro/AppData/Local/npm-cache/_npx/fd45a72a545557e9/node_modules/tsx/dist/cli.mjs','scripts/radio-storage-cycle.mjs']],
 ['build',['node_modules/next/dist/bin/next','build']],
 ['typecheck',['node_modules/typescript/bin/tsc','--noEmit']],
 ['browser',['node_modules/@playwright/test/cli.js','test','-c','tmp/closure-browser.config.ts']],
];
const results=[];
for(const [name,args] of cases){
 if(name==='browser'&&!results.find(r=>r.name==='build'&&r.status==='PASS')){results.push({name,sha,status:'NOT_RUN',reason:'Build incomplete'});continue;}
 const started=new Date().toISOString();const fd=fs.openSync(path.join(out,name+'.log'),'w');
 const r=spawnSync(process.execPath,args,{cwd:root,env,stdio:['ignore',fd,fd],timeout:1200000});fs.closeSync(fd);
 const result={name,sha,command:'node '+args.join(' '),started,finished:new Date().toISOString(),exit:r.status,status:r.status===0?'PASS':'FAIL',error:r.error?.code??null};
 results.push(result);fs.writeFileSync(path.join(out,'results.json'),JSON.stringify(results,null,2)+'\n');console.log(JSON.stringify(result));
}
fs.writeFileSync(path.join(out,'results.json'),JSON.stringify(results,null,2)+'\n');
